$$
\begin{align*}
\phi(z)
&=
\frac{1}{\sqrt{2\pi}}
e^{-z^2/2}
\tag*{\just{The limiting probability density is therefore}}
\\[0.45em]
\int_{-\infty}^{\infty}
e^{-z^2/2}\,dz
&=
\sqrt{2\pi}
\tag*{\just{The Gaussian integral proved that}}
\\[0.45em]
\int_{-\infty}^{\infty}
\phi(z)\,dz
&=
\frac{1}{\sqrt{2\pi}}
\int_{-\infty}^{\infty}
e^{-z^2/2}\,dz
\tag*{\just{Therefore,}}
\\[0.45em]
\int_{-\infty}^{\infty}
\phi(z)\,dz
&=
1
\tag*{\just{Thus,}}
\\[0.45em]
\boxed{
\phi(z)
=
\frac{1}{\sqrt{2\pi}}
e^{-z^2/2}
}
\tag*{\just{Hence,}}
\end{align*}
$$

is a probability density function.

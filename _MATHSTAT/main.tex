\section{Preface}
This handbook is an active draft. I am writing it with two goals in mind: first, to enable the interested reader to learn probability theory from mathematical first principles; second, to bridge what I consider a pedagogical gap between probability theory and its applications.

I originally began writing this as supplementary material for my students, but I am now generalizing it for anyone interested in learning probability theory and its applications—namely statistics and machine learning—from first principles.

\section{Probability Theory}
Probability theory is about mathematical modeling of the phenomena of randomness. In this section, we briefly define fundamental concepts in probability theory.

%----------------------------------------------------------------------------
\subsection{Combinatorics}
Combinatorics is the mathematics of counting things efficiently. In combinatorics, we have techniques for determining the number of possible outcomes of experiments without direct enumeration (i.e, without manually counting). In this section, we will look at some of these techniques which have applications in probability distributions and their functions.

%----------------------------------------------------------------------------
\subsubsection{Permutation with Repetition}
The function for permutation with repetition (or replacement) is given by
$$
P^{r}(n,r) = n^r
$$

%----------------------------------------------------------------------------
\subsubsection{Permutation without Repetition}
The function for permutation without repetition (or replacement) is given by
$$
P(n,r) = nPr = \frac{n!}{(n-r)!}
$$

%----------------------------------------------------------------------------
\subsubsection{Combination without Repetition and the Binomial Coefficient}
The function for combination without repetition (or replacement) is given by
$$
C(n,r) = nCr = \frac{n!}{r!(n-r)!}
$$ 

 
$$
\binom{n}{k} = \frac{n!}{r!(n-r)!}
$$ 

The binomial coefficient notation is used to express combination, alternatively. Binomial coefficients are so called because they are coefficients of terms in the Binomial Theorem.

%----------------------------------------------------------------------------
\subsection{Important Concepts in Probability Theory}
In this section, some properties of probability theory are highlighted because they are relevant throughout the statistics and probability theory.

%----------------------------------------------------------------------------
\subsection{Central Limit Theorem}
The \emph{central limit theorem} states that when samples are drawn from a population, where the size of the samples are 30 or greater, the samples will have a normal distribution. This theorem holds even if the population from which the samples are drawn does is not normally distributed.

%----------------------------------------------------------------------------
\subsection{Law of Large Numbers}
The \emph{law of large numbers} states that as the size of the sample increases, the mean of the sample will approach the mean of its population.

\subsection{Problems}

\begin{itemize}
    \item Establish $\bar{X}_n \xrightarrow{P} \mu$ for independent identically distributed random variables with finite mean.
    \item Establish $\bar{X}_n \xrightarrow{\mathrm{a.s.}} \mu$ for independent identically distributed integrable random variables.
    \item Obtain $S_n/n \xrightarrow{P} 0$ for independent mean-zero summands under weak-law conditions.
    \item Obtain $S_n/n \xrightarrow{\mathrm{a.s.}} 0$ for independent identically distributed mean-zero integrable summands.
    \item Obtain $n^{-1}\sum_{i=1}^n 1_{\{X_i\in A\}} \xrightarrow{\mathrm{a.s.}} P(X\in A)$.
    \item Obtain $n^{-1}\sum_{i=1}^n g(X_i) \xrightarrow{\mathrm{a.s.}} E[g(X)]$ when $g(X)$ is integrable.
    \item Identify $\bar{X}_n \xrightarrow{P} \mu$ as the weak-law conclusion and $\bar{X}_n \xrightarrow{\mathrm{a.s.}} \mu$ as the strong-law conclusion.
\end{itemize}

%----------------------------------------------------------------------------
\subsection{Conditional Probability}
For events $A$ and $B$ with $P(B)>0$, the conditional probability of $A$ given $B$ is

$$
P(A\mid B)=\frac{P(A\cap B)}{P(B)}.
$$

where

\begin{itemize}
    \item $A$ and $B$ are events in the sample space.
    \item $P(A\cap B)$ is the joint probability of $A$ and $B$.
    \item $P(B)$ is the probability of $B$.
    \item $P(A\mid B)$ is the conditional probability of $A$ given $B$.
\end{itemize}

The multiplication rule is

$$
P(A\cap B)=P(A\mid B)P(B)=P(B\mid A)P(A).
$$

The law of total probability states that if $\{B_i\}$ is a partition of the sample space, then

$$
P(A)=\sum_i P(A\mid B_i)P(B_i).
$$

\subsection{Problems}

\begin{itemize}
    \item Obtain $P(A\mid B)=\frac{P(A\cap B)}{P(B)}$ when $P(B)>0$.
    \item Obtain $P(A\cap B)=P(A\mid B)P(B)$.
    \item Obtain $P(A)=\sum_i P(A\mid B_i)P(B_i)$ for a countable partition $\{B_i\}$.
    \item Obtain $P(A_1\cap\cdots\cap A_n)=P(A_1)\prod_{i=2}^n P(A_i\mid A_1\cap\cdots\cap A_{i-1})$.
    \item Obtain $P(A\cap B\mid C)=P(A\mid C)P(B\mid C)$ when $A$ and $B$ are conditionally independent given $C$.
    \item Obtain $P(A^c\mid B)=1-P(A\mid B)$ when $P(B)>0$.
    \item Obtain $P(A\cup C\mid B)=P(A\mid B)+P(C\mid B)-P(A\cap C\mid B)$ when $P(B)>0$.
\end{itemize}

%----------------------------------------------------------------------------
\subsection{Bayes Theorem}
Frequentist statistics treats parameters as fixed and probabilities as long-run frequencies of repeated data.

Bayesian statistics treats unknown parameters and hypotheses probabilistically, combining prior information with observed data to produce posterior probabilities.

Fundamentally, it goes like this:

Frequentist: $P(\text{data}\mid\text{hypothesis})$

Bayesian: $P(\text{hypothesis}\mid\text{data})$

Bayes theorem is given by

$$
P(A\mid B)=\frac{P(B\mid A)P(A)}{P(B)}.
$$

For a partition $\{B_i\}$,

$$
P(B_k\mid A)=\frac{P(A\mid B_k)P(B_k)}{\sum_i P(A\mid B_i)P(B_i)}.
$$

where

\begin{itemize}
    \item $P(A)$ is the prior probability of $A$.
    \item $P(B\mid A)$ is the likelihood of $B$ given $A$.
    \item $P(B)$ is the marginal probability of $B$.
    \item $P(A\mid B)$ is the posterior probability of $A$ given $B$.
    \item $B_k$ is one cell of the partition.
    \item $P(B_k)$ is the prior probability of $B_k$.
    \item $P(A\mid B_k)$ is the likelihood of $A$ given $B_k$.
\end{itemize}

\section{Random Variables}
A \emph{random variable} is a function, mathematically speaking. They map events in a sample space to a subset of the real numbers. But they can also be considered as sets (see §Convergence of Random Variables, Distributions, and Probabilities.) 

A random variable is typically denoted by $X$ and elements in the domain of the random variable denoted $x$; thus $x \in X$. Along with probability distributions, they are central components in Probability Theory.

%----------------------------------------------------------------------------
\subsection{Discrete Random Variables}
A random variable is discrete if it has one of the following two characteristics:
 
\begin{enumerate}
    \item a finite number of possible values
    \item an infinite but countable sequence of possible values (see §Countable Sets.)
\end{enumerate}

%----------------------------------------------------------------------------
\subsection{Continuous Random Variables}
A random variable is continuous if its possible values are infinite and not countable (see §Countable Sets.)

\newpage
\section{Probability Distributions}
A probability distribution is a description of data; in particular, the Observations in the data and their corresponding probability. Probability distributions are analogous to frequency distributions which are descriptions of data and their corresponding Frequencies.

%----------------------------------------------------------------------------
\subsection{Probability Distribution Functions}
Probability distribution functions are models of various types of data. These functions allow us to determine probabilities as functions of Observations and Estimates. Put differently, probability distribution functions will return the probability for a given observation. There are three types of functions which are covered in the following section.

%----------------------------------------------------------------------------
\subsubsection{Probability Mass Function}
The probability mass function of a discrete random variable will map, for every value in the random variable, the probability that the value will be observed. The probability mass function is denoted by,

$$
P(X=k) = p
$$

where $X$ is a discrete random variable, $k$ is an observable value, and $p$ is a probability.

%----------------------------------------------------------------------------
\subsubsection{Probability Density Function}
The probability density function of a continuous random variable will map, for every value in the random variable, the probability that the value will be observed. The probability density function is denoted by,

$$
P(X=k) = p
$$

where $X$ is a continuous random variable, $k$ is an observable value, and $p$ is a probability.

%----------------------------------------------------------------------------
\subsubsection{Cumulative Probability Function}
The cumulative probability function is denoted by

$$
P(X \leq k) = p = \begin{cases} \sum_{i=1}^{k} f(x) \enspace; \quad \text{ if X is Discrete} \\ \int_{-\infty}^{k} f(x) \: dx \enspace; \quad \text{ if X is Continuous} \end{cases}
$$

where 

\begin{itemize}
    \item $X$ is a random variable
    \item $f(x)$ is either a PMF or CDF
    \item $k$ is an observable value
    \item $p$ is a probability
\end{itemize}

The cumulative probability function of a discrete or continuous random variable will map, for every value in the random variable, the probability that the value will be observed.

\section{Overview of Moments}
Moments are specific descriptions of a probability distribution. 

The $kth$ moment is given by
$$
E(X^k)
$$ 

The $kth$ central moment is given by:
$$
E((X-\mu)^k)
$$

%----------------------------------------------------------------------------
\subsection{Expectation}
A probability-weighted average of a random variable that is used to summarize its long-run mean value under a probability model, where a random variable is a measurable function from a sample space to the real line, and where the average is the integral of that function against the probability measure.

Note: Also called expected value or mean.

\begin{mdframed}
\begin{definition}[Expectation of a Discrete Random Variable]
The expectation of a discrete random variable $X$ whose distinct possible values are $x_1,x_2,\ldots$ is

$$
E(X) = \sum_{j} x_j\,P(X=x_j)
$$

where

\begin{itemize}
    \item $X$ is a discrete random variable.
    \item $x_j$ are the distinct values of $X$.
    \item $P(X=x_j)$ is the probability that $X$ equals $x_j$.
\end{itemize}

The expectation is undefined if $\sum_j |x_j|P(X=x_j)$ diverges.
\end{definition}
\end{mdframed}

This principle is used to compute means for countable supports.

\begin{mdframed}
\begin{definition}[Expectation of a Continuous Random Variable]
The expectation of a continuous random variable $X$ with PDF $f$ is

$$
E(X) = \int_{-\infty}^{\infty} x\,f(x)\,dx
$$

where

\begin{itemize}
    \item $X$ is a continuous random variable.
    \item $f$ is the probability density function of $X$.
\end{itemize}

provided that $\int_{-\infty}^{\infty}|x|f(x)\,dx$ is finite.
\end{definition}
\end{mdframed}

This principle is used to compute means for absolutely continuous laws.

\begin{mdframed}
\begin{definition}[Expectation as a Lebesgue Integral]
If $X$ is a random variable on a probability space $(\Omega,\mathcal{F},P)$, the expectation of $X$ is

$$
E(X) = \int_\Omega X\,dP
$$

where

\begin{itemize}
    \item $(\Omega,\mathcal{F},P)$ is a probability space.
    \item $X$ is a random variable on that space.
    \item $dP$ denotes integration with respect to $P$.
\end{itemize}

provided the integral exists. If $E|X|<\infty$, then $X$ is integrable.
\end{definition}
\end{mdframed}

This principle is used to unify discrete and continuous cases as one Lebesgue integral.

Absolute integrability. $X$ is integrable if and only if $|X|$ is integrable, which requires $E|X|<\infty$ and rules out indeterminate $\infty-\infty$ forms. This principle is used to decide when $E(X)$ exists as a real number.

Linearity. For integrable $X,Y$ and constant $c$, $E(X+Y)=E(X)+E(Y)$ and $E(cX)=cE(X)$, even when $X$ and $Y$ are dependent. This principle is used to compute means of sums without joint distributions.

Monotonicity. If $X\le Y$ almost surely, then $E(X)\le E(Y)$, with equality if and only if $X=Y$ almost surely. This principle is used to compare expectations from almost-sure orderings.

Fundamental bridge. For an event $A$ with indicator $I_A$, $P(A)=E(I_A)$. This principle is used to convert probability statements into expectation statements.

\begin{mdframed}
\begin{definition}[Law of the Unconscious Statistician]
LOTUS in the discrete and continuous cases is

$$
E(g(X)) = \sum_x g(x)\,P(X=x), \qquad E(g(X)) = \int_{-\infty}^{\infty} g(x)\,f(x)\,dx
$$

where

\begin{itemize}
    \item $g$ is a real-valued function.
    \item $f$ is the PDF of $X$ in the continuous case.
\end{itemize}
\end{definition}
\end{mdframed}

This principle is used to compute moments of transformed variables without first finding the law of $g(X)$.

Tower property. $E(Y)=E[E(Y\mid X)]$ whenever the expectations exist. This principle is used to compute unconditional means by conditioning.

Markov and Chebyshev. Nonnegative $g$ and $A=\{g(X)\ge a\}$ give $E[g(X)]\ge a\,P(A)$. With $g(x)=|x|$ one obtains Markov’s inequality; with $g(x)=(x-E[X])^2$ one obtains Chebyshev’s inequality. This principle is used to bound tails from moments.

%----------------------------------------------------------------------------
\subsection{Expected Value}
The function of the expected value is given by

$$
E(X) = \begin{cases} \sum_{i}^{k} x_i \cdot f(x_i) \enspace ; \quad \text{ if } X \text{ is discrete} \\ \\ \int_{-\infty}^{\infty} x_i \cdot f(x_i) \: dx_i \enspace ; \quad \text{ if } X \text{ is continuous} \end{cases}
$$ 

where

\begin{itemize}
    \item $x_i$ is a value of a random variable
    \item $f(x_i)$ is a PMF or PDF
\end{itemize}

Expected Value is the average of the values of a random variable. The average is weighted, however, by the probabilities of the outcomes.

%----------------------------------------------------------------------------
\subsection{Variance}
The function for variance is given by
$$
Var(X) = E(X-\mu)^2 = \sigma^2
$$
where 

\begin{itemize}
    \item $X$ is a random variable
    \item $\mu$ is the mean average of the random variable
\end{itemize}

Variance is a measurement of the spread of values of the random variable about the mean. Note that the square root of variance is the standard deviation. In other words, to arrive at the standard deviation of a random variable, the variance must first be calculated. Also, variance can be expressed as

$$
Var(X) = E[X^2] - (E[X])^2
$$
where

\begin{itemize}
    \item $E[X^2] = \begin{cases} \sum_{i}^{k} x_i \cdot f(x_i)^2 \enspace ; \quad \text{ if } X \text{ is discrete} \\ \\ \int_{-\infty}^{\infty} x_i \cdot f(x_i)^2 \: dx_i \enspace ; \quad \text{ if } X \text{ is continuous} \end{cases}$
\end{itemize}

Note, the expression $E[X^2]$ means that the probability function in the function of expected value is squared!

%----------------------------------------------------------------------------
\subsection{Skewness}
The skewness coefficient is given by
$$
\frac{E(X-\mu)^3}{\sigma^3}
$$
where

\begin{itemize}
    \item $E(X-\mu)^3$ is the third moment about the mean. The expression, $(X-\mu)$ is cubed to indicate asymmetry about the mean. If this term were not cubed, the terms in the expression would cancel out and the result would be zero.
    \item $\sigma^3$ is the cube of the standard deviation. This term is in the denominator to make skewness independent of any unit of measurement (e.g., cm, inch, etc.).
\end{itemize}

Methods of Moments are techniques for estimating Parameters of a population. This technique works by equating moments of samples to the theoretical moments of the population from which they are drawn, and then solving for the parameters.

%----------------------------------------------------------------------------
\subsubsection{Moment Generating Functions}
Moment generating functions are used to find the characteristics (i.e, moments) of a probability distribution. Moment generating functions are a type of generating functions. Generating Functions are for representing sequences.

\section{Maximum Likelihood}
Maximum Likelihood is a technique for estimating, by maximizing likelihood, any parameter of a population given the sample.

$$
L(x_1,\dots,x_n ;\theta)=\prod_{i=1}^{n} f(x_i; \theta)
$$ 

where 

\begin{itemize}
    \item $L(x_1, \dots, x_n; \theta)$ is the likelihood function (see next section)
    \item $f(x_i; \theta)$ is the probability distribution function (see Probability Distribution Functions).
    \item $\theta$ is some parameter
    \item $n$ is the number of variables
    \item $ x_1, \dots, x_n $ are variables
    \item $ \prod_{i=1}^{n} $ is the product operator
\end{itemize}

%----------------------------------------------------------------------------
\subsection{Likelihood Function}
The likelihood function is denoted by
$$
L(x_1, \dots, x_n; \theta)
$$
where 

\begin{itemize}
    \item $ x_1, \dots, x_n $ are variables
    \item $\theta$ is the parameter to be estimated
\end{itemize}

Note that the likelihood function can output very small probabilities, which is why it is necessary to apply the logarithm function to the likelihood function (see Log-Likelihood).

%----------------------------------------------------------------------------
\subsection{Log-Likelihood}
The log-likelihood function is the natural logarithm of the likelihood function. log-likelihood function is given by

$$
\ln(L(x_1, \dots, x_n; \theta)) = \ell(x_1, \dots, x_n; \theta) = \log_{e} ( L(x_1, \dots, x_n; \theta) )
$$ 

Note that $ \ln(L(x, \theta)) $ is the natural logarithm; that is, the logarithm of base e. Thus, $ \ln(L(x_1, \dots, x_n, \theta)) = \log_{e} ( L(x_1, \dots, x_n, \theta) ) $.

Also note that capital $L$ is used for likelihood and lowercase italic $l$ for log-likelihood.

Log-likelihood is used, and preferably so, because it "scales" the products of the terms in the MLE — especially when the probabilities are small (e.g, p = $10^{-10} = 0.0000000001 $).

%----------------------------------------------------------------------------
\subsection{Differentiating the Log-Likelihood Function}
Differentiating the log-likelihood function, $\ell(x_1, \dots, x_n,\theta)$, is done to find the critical values of the function. The critical value would then be the value at which the likelihood functions yield the maximum likelihood.

Since the (log) likelihood is a function of multiple variables, we apply the partial derivative to the likelihood function,

$$
\frac{\partial}{\partial x_i} \ell(x_i; \theta) = \frac{\partial}{\partial x} \ln( L(x_i; \theta) )
$$

%----------------------------------------------------------------------------
\subsection{Maximum Likelihood Estimator (MLE)}
An estimator is a function that is used to estimate a parameter.
$$
\hat{\theta}(x_i) = \underset{\theta}{\text{argmax}} \: \ell(x_i; \theta)
$$
where 

\begin{itemize}
    \item $\text{argmax}$ is the function that chooses the argument for which the values of some function is maximized
    \item $ \hat{\theta}(x_i) $ is the maximum likelihood estimator
    \item $ \theta $ is some parameter
    \item $ \ell(x_i; \theta) $ is the log-likelihood function
    \item $ x_i $ is some variable
\end{itemize}

\section{Measure Theory}
Measure Theory is about assigning a notion of size—such as length, area, volume, mass, or probability—to abstract sets, and studying the sets and functions relevant to that notion of size.

%----------------------------------------------------------------------------
\subsection{Measure}
A \emph{measure} is an abstraction of measurable quantities, such as length, weight, volume, and probability.

More formally, a measure is defined as a function that maps a $\sigma$-algebra to $[0,\infty)$. In other words, it maps sets to a real number that is non-negative or infinity. This non-negative property exists because measurable quantities are inherently non-negative.

Usually, in mathematics convention, $\mu$ (pronounced "mu") stands for measure, but since $\mu$ is reserved, we will denote a measure with the uppercase variant, $\mathcal{M}$. 

$$
\mu: S \rightarrow [0,\infty)
$$

%----------------------------------------------------------------------------
\subsubsection{Sigma-algebra}
A \emph{$\sigma$-algebra} or \emph{sigma-algebra} is a collection of subsets. More specifically, it is a type of collection that is closed under \emph{countable unions} and \emph{complements}. A $\sigma$-algebra is necessary, among other operations, for assigning probabilities to subsets of the sample space. The countable unions of the subsets of the sample space have the effect of allowing probabilities to be assigned to events including even infinitely many possible outcomes. A sigma-algebra is also known as a \emph{sigma-field}.

%----------------------------------------------------------------------------
\subsubsection{Measure Space}
A \emph{space} is a set with certain conditions. In mathematics, a space is denoted as a pair of any elements; for example, $(A,B,...)$  We define a measure space together with its conditions formally here.

\begin{mdframed}
\begin{definition}[Measure Space]
A \emph{measure space} is a space $(E, \mathcal{E}, \mathcal{M} )$ where:

\begin{itemize}
    \item $E$ is a set the subsets of which will be measured
    \item $\mathcal{E}$ is a $\sigma$-algebra of measurable subsets of $E$
    \item $\mathcal{M}$ is the measure on $(E, \mathcal{E})$
\end{itemize}
\end{definition}
\end{mdframed}

In this definition, when we say $\mathcal{M}$ is the measure on $(E, \mathcal{E})$, we mean to say $$
\mathcal{M}: \mathcal{E} \rightarrow [0,\infty]
$$.

It is important to indicate that the measure is "on" the pair $(E, \mathcal{E})$ so that we know $\mathcal{E}$ are subsets of $E$.

%----------------------------------------------------------------------------
\subsubsection{Measurable Space}
The following is the definition of the \emph{measurable space}–not to be confused with the "measure space."

\begin{mdframed}
\begin{definition}[Measurable Space]
A \emph{measurable space} is the space $(X,\mathcal{E})$ if

\begin{itemize}
    \item $\mathcal{E} \subset P(X)$
\end{itemize}

where:

\begin{itemize}
    \item $\mathcal{E}$ is $\sigma$-algebra
    \item $X$ is a set
    \item $P(X)$ is the family of all subsets
\end{itemize}
\end{definition}
\end{mdframed}

%----------------------------------------------------------------------------
\subsection{Lebesgue Measure}
The Lebesgue measure, as a special case of a measure, is used for definite integration on areas, volumes, and other higher-dimensional analogues — $n$-dimensional measures, if you will. 

Definite integration on Lebesgue measures is done using Lebesgue Integration. Here, we provide a formal definition.

\begin{mdframed}
\begin{definition}[Lebesgue Meausure]
...

\begin{itemize}
    \item $\mu_F$ is the complete Lebesgue measure
    \item $F(x)=x$ function associated with $\mu_F$
    \item $m$ is the length of the measure
    \item $\mathcal{L}$ is the Lebesgue measurable set
\end{itemize}
\end{definition}
\end{mdframed}

%----------------------------------------------------------------------------
\subsubsection{Lebesgue Integral}
The \emph{lebesgue integral} is a generalization of the Riemann integral.

\begin{mdframed}
\begin{definition}[Lebesgue Integral]
The \emph{Lebesgue integral} is defined by
$$
\int_X f \, d \mathcal{M} = \sup \left\{ \int_X s \, d \mathcal{M} \ \middle|\ s : X \to \mathbb{R} \ \text{ and } 0 \le s \le f \right\}.
$$

where:

\begin{itemize}
    \item $f: X \rightarrow [0,\infty]$ is a non-negative measurable function
    \item $\mathcal{M}$ is the measure
    \item $X$ is a set of the measure space
    \item $s$ is a simple function
\end{itemize}
\end{definition}
\end{mdframed}

%----------------------------------------------------------------------------
\subsection{Probability Measure}
A probability measure is a special case of the general concept of a measure, where the measure of the entire sample space is 1.

More formally, a probability measure is expressed as:

$$
P: \mathcal{F} \rightarrow [0,1]
$$

where 

\begin{enumerate}
    \item $\mathcal{F}$ is a $\sigma$-algebra of subsets of the sample space.
\end{enumerate}

%----------------------------------------------------------------------------
\subsubsection{Probability Space}
A \emph{probability space} is a special case of a measure space which we define here formally.

\begin{mdframed}
\begin{definition}[Probability Space]
A \emph{probability space} is a measure space $(\Omega, \mathcal{F}, \mathbb{P} )$ where:

\begin{itemize}
    \item $\Omega$ is the sample space
    \item $\mathcal{F}$ is the $\sigma$-algebra
    \item $\mathbb{P}$ is the measure with domain $[0,1]$
\end{itemize}
\end{definition}
\end{mdframed}

\section{Markov Chain Monte Carlo}
Markov Chain Monte Carlo is a sampling method. More specfically, it is a class of algorithms for obtaining information about distributions by sampling from them. An example is sampling from the posterior distribution in Bayesian inference.

An MCMC algorithm draws samples from a target distribution by simulating a Markov chain whose long-run distribution is that target distribution.

A Markov chain, in turn, is a sequence of random variables whose next state depends only on the present state, and where a target distribution is the probability distribution one wants to sample from.

Global balance. A distribution $\pi$ is stationary when the probability of each state after one step equals the same probability before the step. This principle is used to write equations that $\pi$ must solve.

\begin{mdframed}
\begin{definition}[Global Balance Equations = Full Balance Equations]
The global balance equations are

\pi_j = \sum_{i \in S} \pi_i p_{ij}, \qquad j \in S

where

\begin{itemize}
    \item $S$ is the discrete state space.
    \item $i$ is a current state.
    \item $j$ is a next state.
    \item $\pi_i$ is the stationary probability of state $i$.
    \item $\pi_j$ is the stationary probability of state $j$.
    \item $p_{ij}$ is the one-step transition probability from $i$ to $j$.
\end{itemize}
\end{definition}
\end{mdframed}

\section{Functional Analysis}
Functional analysis is the study of function spaces and the operators acting on function spaces. In functional analysis, functions are viewed as points in a set.

This idea that functions are treated as points in a set is relevant because this allows us to, in probability theory, guarantee that limits and projections of random variables exist within the sets of functions where probability calculations occur.

\begin{mdblock}
\emph{Example 1.} Let

$$
X_n=\sum_{k=1}^{n}2^{-k}R_k
$$

where

\begin{itemize}
    \item $R_k$ are independent random variables
    \item $E[R_k]=0$
    \item $E[R_k^2]=1$
    \item $2^{-k}$ is the coefficient weighting the $k$th random variable
    \item $k$ is the index of the random variables in the sequence
\end{itemize}

Because $L^2$ is complete,

$$
X_n\xrightarrow{L^2}X.
$$

Here, $X\in L^2$ and $\xrightarrow{L^2}$ denotes convergence in $L^2$.

Because $L^2(\sigma(R_1))$ is a closed subspace,

$$
E[X\mid R_1] = \frac12R_1.
$$

Here, $\sigma(R_1)$ is the sigma-algebra generated by $R_1$.

The conditional expectation exists as the orthogonal projection of $X$ onto $L^2(\sigma(R_1))$ [2, 3].
\end{mdblock}
\subsubsection{\emph{Lemma. $(X_n)$ is Cauchy in $L^2$.}}

Let

$$
\begin{aligned} X_j &= \sum_{k=1}^{j}\left(\frac12\right)^kR_k = 2^{-1}R_1+2^{-2}R_2+\cdots+2^{-j}R_j \\[0.5em] X_m-X_n &= \left(2^{-1}R_1+\cdots+2^{-n}R_n+\cdots+2^{-m}R_m\right) -\left(2^{-1}R_1+\cdots+2^{-n}R_n\right) \\ &= 2^{-(n+1)}R_{n+1}+\cdots+2^{-m}R_m = \sum_{k=n+1}^{m}2^{-k}R_k. \end{aligned}
$$

where $m>n$

\vspace{1\baselineskip}

Therefore, $\|X_m-X_n\|_2^2=E[(X_m-X_n)^2]$.

\vspace{1\baselineskip}

Expanding, $E[(X_m-X_n)^2]=\sum_{k=n+1}^{m}4^{-k}E[R_k^2]$.

\vspace{1\baselineskip}

The cross terms vanish because $R_k$ are independent and $E[R_k]=0$.

\vspace{1\baselineskip}

Since $E[R_k^2]=1$, $\|X_m-X_n\|_2^2=\sum_{k=n+1}^{m}4^{-k}$.

\vspace{1\baselineskip}

Since $\sum_{k=n+1}^{m}4^{-k}\to0$, $\|X_m-X_n\|_2\to0$.

\vspace{1\baselineskip}

Therefore, $(X_n)$ is Cauchy in $L^2$.

\sqrt{2\pi}
\tag*{\just{Solve for the constant}}
\\[0.45em]
\boxed{
n!
\sim
\sqrt{2\pi n}
\left(
\frac{n}{e}
\right)^n
}
&&
\tag*{\just{Stirling's approximation}}
\\[0.45em]
\boxed{
\lim_{n\to\infty}
\frac{
n!
}{
\sqrt{2\pi n}
\left(
\dfrac{n}{e}
\right)^n
}
=
1
}
&&
\tag*{\just{Equivalent limit statement}}
\end{align*}

$\square$

\subsection{Theorem 2: De Moivre-Laplace Theorem}
Theorem. Let

\begin{align*}
X_n
&\sim
\operatorname{Bin}(n,p),
\qquad
0<p<1,
\qquad
q=1-p.
\tag*{\just{Fix a non-degenerate binomial distribution}}
\end{align*}

Then, for fixed real numbers $a<b$,

\begin{align*}
P\left(
a
\leq
\frac{X_n-np}{\sqrt{npq}}
\leq
b
\right)
&\longrightarrow
\int_a^b
\frac{1}{\sqrt{2\pi}}
e^{-z^2/2}
\,dz
\tag*{\just{De Moivre-Laplace limit theorem}}
\end{align*}

as $n\rightarrow\infty$.

Proof.

%----------------------------------------------------------------------------
\subsection{De Moivre-Laplace Approximation}
\begin{align*}
P(X_n=k)
&=
\frac{n!}{k!(n-k)!}p^kq^{n-k}
\tag*{\just{Start from the binomial probability}}
\\[0.45em]
n!
&\approx
\sqrt{2\pi n}
\left(\frac{n}{e}\right)^n
\\
k!
&\approx
\sqrt{2\pi k}
\left(\frac{k}{e}\right)^k
\\
(n-k)!
&\approx
\sqrt{2\pi(n-k)}
\left(\frac{n-k}{e}\right)^{n-k}
\tag*{\just{Approximate each factorial using Theorem 1}}
\\[0.45em]
P(X_n=k)
&\approx
\sqrt{\frac{n}{2\pi k(n-k)}}
\frac{n^np^kq^{n-k}}
{k^k(n-k)^{n-k}}
\tag*{\just{Substitute the three approximations and simplify}}
\\[0.45em]
\delta
&=
k-np
\tag*{\just{Measure the displacement of $k$ from the binomial mean}}
\\[0.45em]
k
&=
np+\delta
\\
n-k
&=
nq-\delta
\tag*{\just{Express both success and failure counts in terms of the displacement}}
\\[0.45em]
L
&=
\log\left[
\frac{n^np^kq^{n-k}}
{k^k(n-k)^{n-k}}
\right]
\tag*{\just{Take the logarithm to convert powers into products}}
\\[0.45em]
&=
-k\log\left(\frac{k}{np}\right)
-(n-k)\log\left(\frac{n-k}{nq}\right)
\tag*{\just{Collect the terms involving $k$ and $n-k$}}
\\[0.45em]
&=
-(np+\delta)
\log\left(1+\frac{\delta}{np}\right)
-(nq-\delta)
\log\left(1-\frac{\delta}{nq}\right)
\tag*{\just{Substitute the displacement expressions}}
\\[0.45em]
\log(1+t)
&=
t-\frac{t^2}{2}+O(t^3)
\\
\log(1-t)
&=
-t-\frac{t^2}{2}+O(t^3)
\tag*{\just{Expand the logarithms for $k$ near the mean}}
\\[0.45em]
L
&\approx
-\frac{\delta^2}{2npq}
\tag*{\just{The first-order terms cancel, leaving the quadratic term}}
\\[0.45em]
\frac{n^np^kq^{n-k}}
{k^k(n-k)^{n-k}}
&\approx
\exp\left(
-\frac{\delta^2}{2npq}
\right)
\tag*{\just{Exponentiate to recover the original factor}}
\\[0.45em]
k
&\approx
np
\\
n-k
&\approx
nq
\tag*{\just{Near the mean, replace the counts by their expected values}}
\\[0.45em]
\sqrt{\frac{n}{2\pi k(n-k)}}
&\approx
\frac{1}{\sqrt{2\pi npq}}
\tag*{\just{Simplify the remaining prefactor}}
\\[0.45em]
P(X_n=k)
&\approx
\frac{1}{\sqrt{2\pi npq}}
\exp\left[
-\frac{(k-np)^2}{2npq}
\right]
\tag*{\just{Combine the prefactor and exponential approximations}}
\\[0.45em]
\mu
&=
np
\\
\sigma^2
&=
npq
\tag*{\just{Apply Proposition 1}}
\\[0.45em]
P(X_n=k)
&\approx
\frac{1}{\sigma\sqrt{2\pi}}
\exp\left[
-\frac{1}{2}
\left(
\frac{k-\mu}{\sigma}
\right)^2
\right]
\tag*{\just{Obtain the normal form of the binomial probability}}
\end{align*}

%----------------------------------------------------------------------------
\subsection{Standardization}
\begin{align*}
z
&=
\frac{k-np}{\sqrt{npq}}
\tag*{\just{Standardize the displacement from the mean}}
\\[0.45em]
&=
\frac{k-\mu}{\sigma}
\tag*{\just{Substitute the binomial mean and standard deviation}}
\\[0.45em]
\Delta k
&=
1
\tag*{\just{$k$ changes in integer increments}}
\\[0.45em]
\Delta z
&=
\frac{1}{\sqrt{npq}}
\tag*{\just{Find the corresponding increment in standardized coordinates}}
\\[0.45em]
P(X_n=k)
&\approx
\frac{1}{\sqrt{2\pi}}
e^{-z^2/2}
\Delta z
\tag*{\just{Write each binomial mass as a Riemann-sum term}}
\\[0.45em]
a
&\leq
\frac{X_n-np}{\sqrt{npq}}
\leq
b
\tag*{\just{Consider a fixed standardized interval}}
\\[0.45em]
P\left(
a
\leq
\frac{X_n-np}{\sqrt{npq}}
\leq
b
\right)
&\approx
\sum
\frac{1}{\sqrt{2\pi}}
e^{-z_i^2/2}
\Delta z
\tag*{\just{Sum the binomial probabilities over the interval}}
\\[0.45em]
n
&\longrightarrow
\infty
\\
\Delta z
&\longrightarrow
0
\tag*{\just{The standardized lattice becomes arbitrarily fine}}
\\[0.45em]
P\left(
a
\leq
\frac{X_n-np}{\sqrt{npq}}
\leq
b
\right)
&\longrightarrow
\int_a^b
\frac{1}{\sqrt{2\pi}}
e^{-z^2/2}
\,dz
\tag*{\just{The Riemann sum converges to the Gaussian integral}}
\end{align*}

\defaultwidthpar{This proves the De Moivre-Laplace theorem.}

$\square$

\subsection{Change of Variables}
\defaultwidthpar{For a one-dimensional integral,}

\begin{align*}
u
&=
g(x)
\tag*{\just{Define a new variable}}
\\[0.45em]
\frac{du}{dx}
&=
g'(x)
\tag*{\just{Differentiate with respect to $x$}}
\\[0.45em]
du
&=
g'(x)\,dx
\tag*{\just{Express the differential}}
\\[0.45em]
\int
f(g(x))g'(x)\,dx
&=
\int
f(u)\,du
\tag*{\just{Substitute $u=g(x)$}}
\\[0.45em]
\int_a^b
f(g(x))g'(x)\,dx
&=
\int_{g(a)}^{g(b)}
f(u)\,du
\tag*{\just{Transform the limits for a definite integral}}
\end{align*}

\defaultwidthpar{The same principle extends to several variables, except that changes in area or volume must also be accounted for.}

%----------------------------------------------------------------------------
\subsection{Jacobian}
Let

\begin{align*}
x
&=
x(u,v)
\\
y
&=
y(u,v)
\tag*{\just{Define a transformation from $(u,v)$ to $(x,y)$}}
\\[0.45em]
J
&=
\begin{pmatrix}
\dfrac{\partial x}{\partial u}
&
\dfrac{\partial x}{\partial v}
\\
\dfrac{\partial y}{\partial u}
&
\dfrac{\partial y}{\partial v}
\end{pmatrix}
\tag*{\just{Form the Jacobian matrix}}
\\[0.45em]
\det J
&=
\frac{\partial x}{\partial u}
\frac{\partial y}{\partial v}
-
\frac{\partial x}{\partial v}
\frac{\partial y}{\partial u}
\tag*{\just{Take its determinant}}
\\[0.45em]
dx\,dy
&=
\left|
\det J
\right|
du\,dv
\tag*{\just{The determinant gives the local area-scaling factor}}
\\[0.45em]
\iint_R
f(x,y)\,dx\,dy
&=
\iint_S
f(x(u,v),y(u,v))
\left|
\det J
\right|
du\,dv
\tag*{\just{Apply the change-of-variables formula}}
\end{align*}

\defaultwidthpar{

where

\begin{itemize}
    \item $R$ is the region in the $(x,y)$ coordinate system.
    \item $S$ is the corresponding region in the $(u,v)$ coordinate system.
    \item $J$ is the Jacobian matrix of the transformation.
\end{itemize}

}

\subsection{Jacobian Example}
\defaultwidthpar{Consider a parallelogram with side vectors}

\begin{align*}
\mathbf{w}_1
&=
\begin{pmatrix}
2\\
1
\end{pmatrix}
\\
\mathbf{w}_2
&=
\begin{pmatrix}
-1\\
1
\end{pmatrix}
\tag*{\just{Choose two side vectors}}
\\[0.45em]
\mathbf{b}
&=
\begin{pmatrix}
1\\
1
\end{pmatrix}
\tag*{\just{Choose the starting point}}
\\[0.45em]
\begin{pmatrix}
x\\
y
\end{pmatrix}
&=
u\mathbf{w}_1
+
v\mathbf{w}_2
+
\mathbf{b}
\tag*{\just{Parameterize every point in the parallelogram}}
\\[0.45em]
x
&=
2u-v+1
\\
y
&=
u+v+1
\tag*{\just{Expand the vector equation}}
\\[0.45em]
0
&\leq
u
\leq
1
\\
0
&\leq
v
\leq
1
\tag*{\just{The unit square maps to the parallelogram}}
\end{align*}

\defaultwidthpar{

The vertices are

\begin{tabularx}{1\textwidth} {| >{\raggedright\arraybackslash}X | >{\raggedright\arraybackslash}X |}
\hline
$(u,v)$ & $(x,y)$ \\
\hline

\begin{align*}
J

\begin{pmatrix}
2 & -1\\

\end{pmatrix}

\\[0.45em]
\det J
&=
(2)(1)-(-1)(1)

3
\tag*{\just{Compute the area-scaling factor}}
\\[0.45em]
A
&=
\iint_P
1\,dx\,dy
\tag*{\just{Express the parallelogram area as an integral}}
\\[0.45em]
&=
\int_0^1
\int_0^1
3\,du\,dv
\tag*{\just{Change variables to the unit square}}
\\[0.45em]
&=
3
\tag*{\just{Evaluate the integral}}
\end{align*}

\defaultwidthpar{This demonstrates why the Jacobian determinant appears in a change of variables: it measures how an area element changes under the transformation.}

\subsection{Polar Coordinates}

\\[0.45em]
J

\begin{pmatrix}

\\
\dfrac{\partial y}{\partial r}
&
\dfrac{\partial y}{\partial\theta}
\end{pmatrix}

\\[0.45em]
&=
\begin{pmatrix}

&

\\

&
r\cos\theta
\end{pmatrix}

\\[0.45em]

\end{align*}

\subsection{Polar Area Element}

\begin{align*}
A

\frac{1}{2}r^2\theta

\\[0.45em]
\frac{\theta}{2\pi}

\\[0.45em]

\frac{\theta}{2\pi}
\pi r^2

\frac{1}{2}r^2\theta
\tag*{\just{Recover the sector-area formula}}
\\[0.45em]
\Delta A_i
&=
\frac{1}{2}r_i^2\Delta\theta
-
\frac{1}{2}r_{i-1}^2\Delta\theta
\tag*{\just{Area between two nearby circular arcs}}
\\[0.45em]

\frac{1}{2}
\left(
r_i^2-r_{i-1}^2
\right)
\Delta\theta
\tag*{\just{Factor out the angular width}}
\\[0.45em]

\left(
r_i-r_{i-1}
\right)
\Delta\theta
\tag*{\just{Factor the difference of squares}}
\\[0.45em]
r_i^*

\end{align*}

\subsection{Lemma 1: Gaussian Integral}
Lemma. The Gaussian integral satisfies

\begin{align*}
\int_{-\infty}^{\infty}
e^{-x^2/2}\,dx
&=
\sqrt{2\pi}
\tag*{\just{Gaussian integral}}
\end{align*}

Proof.

\begin{align*}
I
&=
\int_{-\infty}^{\infty}
e^{-x^2/2}\,dx
\tag*{\just{Define the Gaussian integral}}
\\[0.45em]
I^2
&=
\left(
\int_{-\infty}^{\infty}
e^{-x^2/2}\,dx
\right)
\left(
\int_{-\infty}^{\infty}
e^{-y^2/2}\,dy
\right)
\tag*{\just{Square the integral}}
\\[0.45em]
&=
\int_{-\infty}^{\infty}
\int_{-\infty}^{\infty}
e^{-x^2/2}
e^{-y^2/2}
\,dx\,dy
\tag*{\just{Write the product as an integral over the plane}}
\\[0.45em]
e^ae^b
&=
e^{a+b}
\tag*{\just{Use the product rule for exponentials}}
\\[0.45em]
I^2
&=
\int_{-\infty}^{\infty}
\int_{-\infty}^{\infty}
e^{-(x^2+y^2)/2}
\,dx\,dy
\tag*{\just{Combine the exponential factors}}
\\[0.45em]
x
&=
r\cos\theta
\\
y
&=
r\sin\theta
\\
x^2+y^2
&=
r^2
\tag*{\just{Switch to polar coordinates}}
\\[0.45em]
dx\,dy
&=
r\,dr\,d\theta
\tag*{\just{Apply the polar Jacobian}}
\\[0.45em]
I^2
&=
\int_0^{2\pi}
\int_0^\infty
e^{-r^2/2}
r\,dr\,d\theta
\tag*{\just{Transform the integral over the entire plane}}
\\[0.45em]
u
&=
\frac{r^2}{2}
\\
du
&=
r\,dr
\tag*{\just{Substitute in the radial integral}}
\\[0.45em]
\int_0^\infty
e^{-r^2/2}
r\,dr
&=
\int_0^\infty
e^{-u}\,du
\tag*{\just{Transform the inner integral}}
\\[0.45em]
&=
\left[
-e^{-u}
\right]_0^\infty
\\
&=
1
\tag*{\just{Evaluate the inner integral}}
\\[0.45em]
I^2
&=
\int_0^{2\pi}
1\,d\theta
\\
&=
2\pi
\tag*{\just{Evaluate the angular integral}}
\\[0.45em]
I
&=
\sqrt{2\pi}
\tag*{\just{Take the positive square root because $I>0$}}
\end{align*}

Therefore,

\begin{align*}
\boxed{
\int_{-\infty}^{\infty}
e^{-x^2/2}\,dx
=
\sqrt{2\pi}
}
&&
\tag*{\just{Gaussian integral}}
\\[0.45em]
\boxed{
\int_{-\infty}^{\infty}
e^{-x^2}\,dx
=
\sqrt{\pi}
}
&&
\tag*{\just{Equivalent scaled form}}
\end{align*}

$\square$

\subsection{Corollary 5: Variance}
Corollary. If

\tag*{\just{Assume a general normal random variable}}

\\[0.45em]
E[Z^2]
&=
\frac{1}{\sqrt{2\pi}}
\int_{-\infty}^{\infty}
z^2e^{-z^2/2}\,dz
\tag*{\just{Write the second moment}}
\\[0.45em]
u
&=
z
\\
dv
&=
z e^{-z^2/2}\,dz
\tag*{\just{Choose terms for integration by parts}}
\\[0.45em]
du
&=
dz
\\
v
&=
-e^{-z^2/2}
\tag*{\just{Differentiate and integrate the chosen terms}}
\\[0.45em]
\int_{-\infty}^{\infty}
z^2e^{-z^2/2}\,dz
&=
\left[
-z e^{-z^2/2}
\right]_{-\infty}^{\infty}
+
\int_{-\infty}^{\infty}
e^{-z^2/2}\,dz
\tag*{\just{Apply integration by parts}}
\\[0.45em]
&=
\sqrt{2\pi}
\tag*{\just{The boundary term vanishes and Lemma 1 evaluates the integral}}
\\[0.45em]
E[Z^2]
&=
1
\tag*{\just{Divide by $\sqrt{2\pi}$}}
\\[0.45em]
\operatorname{Var}(Z)
&=
1
\tag*{\just{Standard normal variance}}
\\[0.45em]
X
&=
\mu+\sigma Z
\tag*{\just{Use Definition 3}}
\\[0.45em]
\operatorname{Var}(X)
&=
\sigma^2
\operatorname{Var}(Z)
\tag*{\just{Variance is invariant under translation and quadratic under scaling}}
\\[0.45em]
&=
\sigma^2
\tag*{\just{Substitute $\operatorname{Var}(Z)=1$}}
\\[0.45em]
\boxed{
\operatorname{Var}(X)
=
\sigma^2
}
&&
\tag*{\just{Variance of the general normal distribution}}
\end{align*}

$\square$

%----------------------------------------------------------------------------
\subsection{Final Result}
\begin{align*}
X
&\sim
N(\mu,\sigma^2)
\tag*{\just{General normal random variable}}
\\[0.45em]
f(x;\mu,\sigma)
&=
\frac{1}{\sigma\sqrt{2\pi}}
\exp\left[
-\frac{1}{2}
\left(
\frac{x-\mu}{\sigma}
\right)^2
\right]
\tag*{\just{Probability density}}
\\[0.45em]
\int_{-\infty}^{\infty}
f(x;\mu,\sigma)\,dx
&=
1
\tag*{\just{Normalization}}
\\[0.45em]
E[X]
&=
\mu
\tag*{\just{Mean}}
\\[0.45em]
\operatorname{Var}(X)
&=
\sigma^2
\tag*{\just{Variance}}
\end{align*}

\defaultwidthpar{The normal density therefore arises as the continuous limiting form of the binomial distribution under standardization, with the Gaussian integral supplying its normalization.}
